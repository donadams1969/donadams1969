        require(blockedAddresses[_addr], "SGAU: Not blocked");

        require(!blockedAddresses[_addr], "SGAU: Blocked address");

        require(approvedOperators[_addr], "SGAU: Not operator");
        approvedOperators[_addr] = false;
        emit OperatorRevoked(_addr, block.timestamp);

pragma solidity ^0.8.19;

import "@openzeppelin/contracts/utils/cryptography/ECDSA.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

/// @title SGAU_SignatureVerifier
/// @author Commander DG77.77X-Ξ
/// @notice ECDSA signature + nonce replay protection
contract SGAU_SignatureVerifier is Ownable {
    using ECDSA for bytes32;

    address public trustedSigner;
    mapping(bytes32 => bool) public usedNonces;
    mapping(address => uint256) public verificationCount;

    event SignatureVerified(address indexed signer, bytes32 indexed messageHash, bytes32 nonce, uint256 timestamp);
    event TrustedSignerUpdated(address indexed oldSigner, address indexed newSigner);

    constructor(address _trustedSigner) Ownable(msg.sender) {
        require(_trustedSigner != address(0), "SGAU: Invalid signer");
        trustedSigner = _trustedSigner;
    }

    function verifySignature(
        bytes32 messageHash,
        bytes memory signature,
        bytes32 nonce
    ) public returns (bool) {
        require(!usedNonces[nonce], "SGAU: Nonce used");
        require(signature.length == 65, "SGAU: Invalid sig length");

        address recovered = messageHash.toEthSignedMessageHash().recover(signature);
        require(recovered == trustedSigner, "SGAU: Invalid signature");

        usedNonces[nonce] = true;
        verificationCount[recovered]++;

        emit SignatureVerified(recovered, messageHash, nonce, block.timestamp);
        return true;
    }

    function updateTrustedSigner(address _newSigner) external onlyOwner {
        require(_newSigner != address(0), "SGAU: Invalid signer");
        emit TrustedSignerUpdated(trustedSigner, _newSigner);
        trustedSigner = _newSigner;
    }

    function isNonceUsed(bytes32 nonce) external view returns (bool) {
        return usedNonces[nonce];
    }
}

## Contract 3 – SGAU_MultiSig.sol

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.19;

/// @title SGAU_MultiSig
/// @author Commander DG77.77X-Ξ
/// @notice Multi-owner threshold execution
contract SGAU_MultiSig {
    address[] public owners;
    uint256 public required;
    mapping(uint256 => Transaction) public transactions;
    mapping(uint256 => mapping(address => bool)) public confirmations;
    mapping(address => bool) public isOwner;
    uint256 public transactionCount;

    struct Transaction {
        address destination;
        uint256 value;
        bytes data;
        bool executed;
        uint256 confirmationCount;
    }

    event Submission(uint256 indexed transactionId, address indexed submitter);
    event Confirmation(address indexed owner, uint256 indexed transactionId);
    event Execution(uint256 indexed transactionId);
    event ExecutionFailure(uint256 indexed transactionId);

    modifier onlyOwner() { require(isOwner[msg.sender], "SGAU: Not owner"); _; }
    modifier transactionExists(uint256 id) { require(id < transactionCount, "SGAU: No tx"); _; }
    modifier notExecuted(uint256 id) { require(!transactions[id].executed, "SGAU: Executed"); _; }
    modifier notConfirmed(uint256 id) { require(!confirmations[id][msg.sender], "SGAU: Confirmed"); _; }

    constructor(address[] memory _owners, uint256 _required) {
        require(_owners.length > 0 && _required > 0 && _required <= _owners.length, "SGAU: Invalid");
        for (uint i = 0; i < _owners.length; i++) {
            address owner = _owners[i];
            require(owner != address(0) && !isOwner[owner], "SGAU: Invalid owner");
            isOwner[owner] = true;
            owners.push(owner);
        }
        required = _required;
    }

    function submitTransaction(address dest, uint256 val, bytes memory data) public onlyOwner returns (uint256) {
        uint256 id = transactionCount++;
        transactions[id] = Transaction(dest, val, data, false, 0);
        emit Submission(id, msg.sender);
        confirmTransaction(id);
        return id;
    }

    function confirmTransaction(uint256 id) public onlyOwner transactionExists(id) notConfirmed(id) {
        confirmations[id][msg.sender] = true;
        transactions[id].confirmationCount++;
        emit Confirmation(msg.sender, id);
        if (transactions[id].confirmationCount >= required) executeTransaction(id);
    }

    function executeTransaction(uint256 id) public onlyOwner transactionExists(id) notExecuted(id) {
        Transaction storage txn = transactions[id];
        if (txn.confirmationCount >= required) {
            txn.executed = true;
            (bool success, ) = txn.destination.call{value: txn.value}(txn.data);
            success ? emit Execution(id) : emit ExecutionFailure(id);
        }
    }

    receive() external payable {}
}
